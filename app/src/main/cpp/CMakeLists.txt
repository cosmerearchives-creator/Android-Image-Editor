cmake_minimum_required(VERSION 3.22.1)
project(android_image_editor_xr LANGUAGES CXX)
find_package(OpenXR REQUIRED CONFIG)
add_library(image_editor_xr SHARED vr_native.cpp)
target_compile_features(image_editor_xr PRIVATE cxx_std_17)
target_compile_definitions(image_editor_xr PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
target_link_libraries(image_editor_xr PRIVATE OpenXR::openxr_loader EGL GLESv3 log android)
